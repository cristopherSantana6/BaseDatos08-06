-- 3. Crear una tabla llamada TDepartamento.
CREATE TABLE TDepartamento (
    nDepartamentoID INT IDENTITY(1,1),
    cNombreDepartamento VARCHAR(100) NOT NULL,
    -- Definición de restricciones
    CONSTRAINT PK_TDepartamento PRIMARY KEY (nDepartamentoID),
    CONSTRAINT UQ_cNombreDepartamento UNIQUE (cNombreDepartamento)
);
GO

-- 4. Crear una tabla llamada TCargo.
CREATE TABLE TCargo (
    nCargoID INT IDENTITY(1,1),
    cNombreCargo VARCHAR(100) NOT NULL,
    -- Definición de restricciones
    CONSTRAINT PK_TCargo PRIMARY KEY (nCargoID),
    CONSTRAINT UQ_cNombreCargo UNIQUE (cNombreCargo)
);
GO

-- 5. Crear una tabla llamada TEmpleado.
-- Nota: Se incluyen las llaves foráneas (FK) correspondientes para mantener la integridad referencial con TDepartamento y TCargo.
CREATE TABLE TEmpleado (
    nEmpleadoID INT IDENTITY(1,1),
    cNIF VARCHAR(20) NULL, -- Se asume que permite nulos pero los valores ingresados deben ser únicos
    cNombre VARCHAR(50) NOT NULL,
    cApellido VARCHAR(50) NOT NULL,
    nDepartamentoID INT NULL,
    nCargoID INT NULL,
    dFechaContratacion DATE NULL,
    nSalario DECIMAL(18, 2) NULL,
    -- Definición de restricciones
    CONSTRAINT PK_TEmpleado PRIMARY KEY (nEmpleadoID),
    CONSTRAINT UQ_cNIF UNIQUE (cNIF),
    CONSTRAINT FK_TEmpleado_TDepartamento FOREIGN KEY (nDepartamentoID) 
        REFERENCES TDepartamento(nDepartamentoID) ON DELETE SET NULL,
    CONSTRAINT FK_TEmpleado_TCargo FOREIGN KEY (nCargoID) 
        REFERENCES TCargo(nCargoID) ON DELETE SET NULL
);
GO

-- 15. Crear tabla intermedia TEmpleadoProyecto para relación muchos a muchos.
-- Esta tabla vincula TEmpleado y TProyecto, y su llave primaria es compuesta por ambos IDs.
CREATE TABLE TEmpleadoProyecto (
    nEmpleadoID INT NOT NULL,
    nProyectoID INT NOT NULL,
    dFechaAsignacion DATE NULL CONSTRAINT DF_dFechaAsignacion DEFAULT GETDATE(), -- Campo extra opcional para auditoría
    
    -- Definición de la Clave Primaria Compuesta
    CONSTRAINT PK_TEmpleadoProyecto PRIMARY KEY (nEmpleadoID, nProyectoID),
    
    -- Llaves foráneas que garantizan la integridad de la relación muchos a muchos
    CONSTRAINT FK_TEmpleadoProyecto_TEmpleado FOREIGN KEY (nEmpleadoID) 
        REFERENCES TEmpleado(nEmpleadoID) ON DELETE CASCADE,
        
    CONSTRAINT FK_TEmpleadoProyecto_TProyecto FOREIGN KEY (nProyectoID) 
        REFERENCES TProyecto(nProyectoID) ON DELETE CASCADE
);
GO
